cmake_minimum_required(VERSION 3.20)
project(WeFlowLocalProvider LANGUAGES C CXX)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(TENCENT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/wcdb-2.1.16")
add_subdirectory("${TENCENT_ROOT}/src" tencent)
add_library(weflow_wcdb_provider SHARED provider.cpp hello.cpp)
target_compile_features(weflow_wcdb_provider PRIVATE cxx_std_17)
target_compile_options(weflow_wcdb_provider PRIVATE /utf-8 /EHsc)
configure_file("${TENCENT_ROOT}/sqlcipher/sqlite3.h" "${CMAKE_CURRENT_SOURCE_DIR}/include/sqlite3.h" COPYONLY)
target_include_directories(weflow_wcdb_provider PRIVATE include)
target_link_directories(weflow_wcdb_provider PRIVATE "${TENCENT_ROOT}/tools/prebuild/openssl/windows/win64")
target_link_libraries(weflow_wcdb_provider PRIVATE sqlcipher zstd libcrypto Crypt32 Ws2_32 bcrypt windowsapp)
set_target_properties(weflow_wcdb_provider PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../../resources/local-wcdb/win32/x64")
add_executable(create_fixture fixture.cpp)
target_compile_features(create_fixture PRIVATE cxx_std_17)
target_compile_options(create_fixture PRIVATE /utf-8 /EHsc)
target_include_directories(create_fixture PRIVATE include)
target_link_directories(create_fixture PRIVATE "${TENCENT_ROOT}/tools/prebuild/openssl/windows/win64")
target_link_libraries(create_fixture PRIVATE sqlcipher zstd libcrypto Crypt32 Ws2_32)
add_executable(weflow_local_export local_export.cpp)
target_compile_features(weflow_local_export PRIVATE cxx_std_17)
target_compile_options(weflow_local_export PRIVATE /utf-8 /EHsc)
target_include_directories(weflow_local_export PRIVATE include)
target_link_libraries(weflow_local_export PRIVATE weflow_wcdb_provider)
set_target_properties(weflow_local_export PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/../../resources/local-wcdb/win32/x64")
